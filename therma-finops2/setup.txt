#!/bin/bash

echo "Starting Therma-FinOps EC2 Setup with Audit Logging (IST Timezone)..."

# 1. Update packages and install Docker
echo "Installing Docker..."
sudo yum update -y
sudo yum install docker -y
sudo service docker start

# 2. Grant Docker permissions to ec2-user
sudo usermod -a -G docker ec2-user

# 3. Create the working directories (including logs)
echo "Creating project files and log directories..."
mkdir -p ~/therma-worker/logs
chmod 777 ~/therma-worker/logs
cd ~/therma-worker

# 4. Generate the Python worker script with Logging (IST)
cat << 'EOF' > run_worker.py
import os
import time
import math
from datetime import datetime, timedelta, timezone

task_id = os.getenv("TASK_ID", "unknown")
print(f"Cloud container executing Task: {task_id}")

# Simulate heavy CPU workload
end_time = time.time() + 5
while time.time() < end_time:
    math.factorial(10000)

print(f"Task {task_id} completed successfully in cloud.")

# --- IST Timezone Fix ---
# Create a timezone object for UTC + 5:30 (Indian Standard Time)
ist_timezone = timezone(timedelta(hours=5, minutes=30))
current_time = datetime.now(ist_timezone).strftime('%Y-%m-%d %H:%M:%S')

# Write to a permanent audit log
log_entry = f"[{current_time}] SUCCESS: {task_id} executed on AWS Cloud Node.\n"
with open('/app/logs/finops_audit.log', 'a') as f:
    f.write(log_entry)
EOF

# 5. Generate the Dockerfile
cat << 'EOF' > Dockerfile
FROM python:3.9-slim
WORKDIR /app
COPY run_worker.py .
RUN mkdir -p /app/logs
CMD ["python", "run_worker.py"]
EOF

# 6. Build the Docker Image
echo "Building the Docker image..."
sudo docker 
build -t therma-finops-worker:latest .

echo "=========================================================="
echo "✅ Setup Complete! The Therma-FinOps worker is ready."
echo "⚠️  REMINDER: Update your local Windows .env file with"
echo "   today's new AWS_EC2_IP address before running main.py."
echo "=========================================================="
